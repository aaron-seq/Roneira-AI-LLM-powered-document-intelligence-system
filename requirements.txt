# --- Authentication ----------------------------------------------------
# PyJWT rather than python-jose. The pinned python-jose 3.3.0 carried
# CVE-2024-33663 (algorithm confusion with an ECDSA public key) and
# CVE-2024-33664 (a JWE token that decompresses without bound); upgrading it
# to 3.5.0 fixed those but still pulled in `ecdsa`, which carries
# CVE-2024-23342 (Minerva timing attack on P-256) with *no fixed version* —
# its maintainers do not treat side-channel resistance as in scope.
#
# This service signs with HS256, so it never executes an ECDSA path; carrying
# an unfixable advisory for an unused code path is noise in every audit. PyJWT
# is the de-facto standard, actively maintained, and offers the same
# encode/decode surface — the migration was three lines in auth_service.py.
# The [crypto] extra preserves RS*/ES* support for anyone who sets ALGORITHM.
PyJWT[crypto]==2.13.0
# Only pbkdf2_sha256 is used, which is pure Python inside passlib and needs no
# compiled bcrypt backend — see backend/services/auth_service.py.
passlib==1.7.4

# --- Retrieval ---------------------------------------------------------
# chromadb 0.5.0 used `np.float_`, removed in NumPy 2, which forced `numpy<2`.
# NumPy 1.26 in turn has no CPython 3.13 wheels, so the pair capped the whole
# project at Python 3.12. chromadb 1.x is NumPy 2 clean and keeps the client
# surface this repository uses: PersistentClient, get_or_create_collection,
# add/query/get/delete/count (see backend/services/vector_store_service.py).
#
# `pip-audit` reports four open advisories against every chromadb release to
# date, with no fixed version available. All four are against Chroma's *HTTP
# server*: CVE-2026-45829 and CVE-2026-45833 are code injection through
# `trust_remote_code` on the /api/v2/.../collections endpoints, CVE-2026-45830
# and CVE-2026-45831 are missing tenant checks in its authorization providers.
# This service embeds Chroma in-process via PersistentClient — no HTTP
# listener, no auth provider, no tenants — so none of the four are reachable.
# Three of them also affected the 0.5.0 pin this replaced. Re-check when a
# fixed release lands; the CI dependency audit is advisory for this reason.
chromadb==1.5.9
numpy>=2.0
# Supplies both the bi-encoder used for embeddings and the cross-encoder used
# for reranking. Optional at runtime: without it search degrades to keyword
# matching and says so.
sentence-transformers>=3.0.0
